select log_game, count(1) as qa, sum(amount) as ga from es_billing where created_at>='2025-11-06 00:00:00' and created_at<='2025-11-06 23:59:59' group by log_game order BY log_game
select item_id, item_name, sum(item_qty) as qa, sum(final_gross_amount) as da from  es_customer_direct_sale_detail where created_at>='2025-11-06 00:00:00' and created_at<='2025-11-06 23:59:59'  group by item_id, item_name order by item_name, item_id
