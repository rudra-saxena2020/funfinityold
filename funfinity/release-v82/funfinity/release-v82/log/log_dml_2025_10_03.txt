CREATE FUNCTION `get_used_item_qty`(`input_item_id` INT) RETURNS INT READS SQL DATA
BEGIN
    DECLARE output_es_customer_direct_qty_new INT;
    DECLARE output_es_customer_direct_qty INT;
    DECLARE output_es_customer_token_gift_qty INT;

    SELECT IFNULL(SUM(item_qty), 0) INTO output_es_customer_direct_qty_new FROM es_customer_direct_sale_detail  WHERE item_id = input_item_id;    
    SELECT IFNULL(SUM(1), 0) INTO output_es_customer_token_gift_qty FROM es_customer_token_gift WHERE item_id = input_item_id;

    RETURN output_es_customer_direct_qty_new /*+ output_es_customer_direct_qty*/ + output_es_customer_token_gift_qty;
END;

CREATE FUNCTION `get_used_item_qty`(`input_item_id` INT) RETURNS INT READS SQL DATA
BEGIN
    DECLARE output_es_customer_direct_qty_new INT;
    DECLARE output_es_customer_direct_qty INT;
    DECLARE output_es_customer_token_gift_qty INT;

    SELECT IFNULL(SUM(item_qty), 0) INTO output_es_customer_direct_qty_new FROM es_customer_direct_sale_detail  WHERE item_id = input_item_id;    
    SELECT IFNULL(SUM(1), 0) INTO output_es_customer_token_gift_qty FROM es_customer_token_gift WHERE item_id = input_item_id;

    RETURN output_es_customer_direct_qty_new /*+ output_es_customer_direct_qty*/ + output_es_customer_token_gift_qty;
END;
