select log_game, count(1) as qa, sum(amount) as ga from es_billing where created_at>='2026-02-12 00:00:00' and created_at<='2026-02-12 23:59:59' group by log_game order BY log_game
select item_id, item_name, sum(item_qty) as qa, sum(final_gross_amount) as da from  es_customer_direct_sale_detail where created_at>='2026-02-12 00:00:00' and created_at<='2026-02-12 23:59:59'  group by item_id, item_name order by item_name, item_id
SELECT recharge_amount , COUNT(recharge_amount) as c from es_payment WHERE recharge_amount IN (SELECT amount FROM es_bonus) and created_at>='2026-02-12 00:00:00' and created_at<='2026-02-12 23:59:59'  GROUP BY recharge_amount ORDER BY recharge_amount;
SELECT CASE WHEN ec.created_at = ep.created_at THEN 'NEW' ELSE 'OLD' END AS card_type, COUNT(*) AS payment_count, SUM(ep.recharge_amount) AS total_amount FROM es_payment AS ep INNER JOIN es_customer AS ec ON ec.guid = ep.guid AND ec.rfid = ep.rfid WHERE ec.is_vip = 1 and ep.created_at>='2026-02-12 00:00:00' and ep.created_at<='2026-02-12 23:59:59'  GROUP BY CASE WHEN ec.created_at = ep.created_at THEN 'NEW' ELSE 'OLD' END
