select _ep.id as `Payment ID`, _ep.rfid as `RFID`, GUID, _ep.name as `Name`, _ep.phone as `Phone`, _ep.payment_mode as `Payment Mode`, _ep.pre_amount as `Pre Amount`, _ep.recharge_amount as `Recharge Amount`, _ep.discount_amount_on_bill as `Discount Amount`, (_ep.recharge_amount - _ep.discount_amount_on_bill) as `Collect Amount`, ROUND( _ep.collect_amount_from_customer - _ep.gst_amount ,2 ) as `Taxable Amount`, ROUND(_ep.gst_amount / 2 ,  2) as `CGST Amount`, ROUND(_ep.gst_amount / 2 ,  2) as `SGST Amount`, _ep.main_amount as `Main Amount`, _ep.bonus_amount as `Bonus Amount`, _ep.post_amount as `Post Amount`, _ep.user as `User`, _ep.created_at as `Date` from es_payment as _ep where _ep.created_at>='2026-01-18 00:00:01' and _ep.created_at<='2026-01-18 23:59:59'  and _ep.is_deleted <> 'yes' order by _ep.id desc
